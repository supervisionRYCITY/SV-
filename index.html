/** * SuperVision+ Backend API * เพิ่มการรองรับข้อมูลข่าวประชาสัมพันธ์ */ function doGet(e) { var action = e ? e.parameter.action : ''; if (action === 'getAllData') { return createJsonResponse(getAllData()); } return createJsonResponse({ status: 'error', message: 'Invalid Action' }); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; if (action === 'login') { return createJsonResponse(handleLogin(data.username, data.password)); } if (action === 'saveData') { return createJsonResponse(saveData(data.sheetName, data.rowObj)); } if (action === 'deleteData') { return createJsonResponse(deleteData(data.sheetName, data.id)); } return createJsonResponse({ status: 'error', message: 'Unknown Action' }); } catch (err) { return createJsonResponse({ status: 'error', message: err.toString() }); } } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function sheetToObjects(sheet) { if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var rows = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[0] === "" || row[0] === null || row[0] === undefined) continue; var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = row[j]; } rows.push(obj); } return rows; } function getAllData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return { status: 'success', users: sheetToObjects(ss.getSheetByName('ผู้ใช้งาน')), units: sheetToObjects(ss.getSheetByName('ข้อมูลหน่วยงาน')), students: sheetToObjects(ss.getSheetByName('ข้อมูลนักเรียน')), personnel: sheetToObjects(ss.getSheetByName('ข้อมูลบุคลากร')), news: sheetToObjects(ss.getSheetByName('ข่าวประชาสัมพันธ์')) // ดึงข้อมูลข่าวสารเพิ่ม }; } function handleLogin(username, password) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var users = sheetToObjects(ss.getSheetByName('ผู้ใช้งาน')); var user = users.find(function(u) { return String(u['Username']) === String(username) && String(u['Password']) === String(password); }); if (user) { return { status: 'success', user: { name: user['ชื่อผู้ใช้งาน'], role: user['สิทธิ์การใช้งาน'], position: user['ตำแหน่ง'], unit: user['หน่วยงาน'] } }; } else { return { status: 'error', message: 'ชื่อผู้ใช้งานหรือรหัสผ่านไม่ถูกต้อง' }; } } function saveData(sheetName, rowObj) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); // ถ้ายังไม่มีชีท ข่าวประชาสัมพันธ์ ให้สร้างอัตโนมัติ if (!sheet && sheetName === 'ข่าวประชาสัมพันธ์') { sheet = ss.insertSheet('ข่าวประชาสัมพันธ์'); sheet.appendRow(['ลำดับ', 'พาดหัวข่าว', 'เนื้อหาข่าว', 'ภาพข่าว', 'วันที่']); } if (!sheet) return { status: 'error', message: 'ไม่พบชีท: ' + sheetName }; var data = sheet.getDataRange().getValues(); var headers = data[0]; var targetRowIndex = -1; var id = rowObj['ลำดับ']; if (id !== undefined && id !== '' && id !== null) { for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id)) { targetRowIndex = i + 1; break; } } } var rowValues = []; headers.forEach(function(h) { rowValues.push(rowObj[h] !== undefined ? rowObj[h] : ''); }); if (targetRowIndex > 0) { sheet.getRange(targetRowIndex, 1, 1, headers.length).setValues([rowValues]); } else { var maxId = 0; for (var j = 1; j < data.length; j++) { var currentId = Number(data[j][0]); if (!isNaN(currentId) && currentId > maxId) { maxId = currentId; } } rowValues[0] = maxId + 1; sheet.appendRow(rowValues); } return { status: 'success' }; } function deleteData(sheetName, id) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) return { status: 'error', message: 'ไม่พบชีท' }; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id)) { sheet.deleteRow(i + 1); return { status: 'success' }; } } return { status: 'error', message: 'ไม่พบรายการที่ต้องการลบ' }; }
